data.predicted_techniques.join(", ") : "No clear next technique predicted"; killChainTime.textContent = data.time_estimate ? `Estimated time to next stage: ${data.time_estimate}` : ""; killChainActions.innerHTML = ""; (data.recommended_actions || []).slice(0, 3).forEach((action) => { const item = document.createElement("li"); item.textContent = action; killChainActions.appendChild(item); }); } fetch("/api/killchain/{{ incident.src_ip }}/") .then((response) => response.json()) .then((data) => renderKillChain(data)) .catch(() => { killChainStatus.textContent = "Kill chain analysis unavailable."; killChainCurrent.textContent = "No confirmed stage yet"; killChainConfidence.textContent = ""; killChainTechniques.textContent = "-"; }); if (generateButton) { generateButton.addEventListener("click", async () => { generateButton.disabled = true; analysisStatus.textContent = "Analyzing..."; try { const response = await fetch("/api/incidents/{{ incident.id }}/summary/", { method: "POST", headers: { "X-CSRFToken": getCsrfToken(), "Content-Type": "application/json", }, }); const payload = await response.json(); if (!response.ok || !payload.summary) { throw new Error(payload.error || "Analysis request failed"); } window.location.reload(); } catch (error) { generateButton.disabled = false; analysisStatus.textContent = error.message; analysisStatus.style.color = "var(--danger)"; } }); }
